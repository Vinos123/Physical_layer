cmake_minimum_required(VERSION 3.16)
project(nr_pusch_learning_path LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(NR_USE_SOLUTIONS "Build reference solutions instead of student.cpp" OFF)
add_library(nr_reference ../cpp/src/resource.cpp ../cpp/src/coding.cpp ../cpp/src/phy.cpp ../cpp/src/channel.cpp)
target_include_directories(nr_reference PUBLIC ../cpp/include)
if(NR_USE_SOLUTIONS)
  set(NR_LAB_SOURCE exercises/solutions.cpp)
else()
  set(NR_LAB_SOURCE exercises/student.cpp)
endif()
add_executable(nr_labs exercises/tests.cpp ${NR_LAB_SOURCE})
target_link_libraries(nr_labs PRIVATE nr_reference)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(nr_labs PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
foreach(NR_LAB_ID L01 L02 L03 L04 L05 L06 L07 L08 L09 L10)
  add_test(NAME ${NR_LAB_ID} COMMAND nr_labs ${NR_LAB_ID})
endforeach()
