cmake_minimum_required(VERSION 3.16)
project(pusch_learning_lab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(PUSCH_SANITIZE "Enable address and undefined-behaviour sanitizers" OFF)
add_library(pusch_lab src/resource.cpp src/coding.cpp src/phy.cpp src/channel.cpp)
target_include_directories(pusch_lab PUBLIC include)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(pusch_lab PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
  if(PUSCH_SANITIZE)
    target_compile_options(pusch_lab PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(pusch_lab PUBLIC -fsanitize=address,undefined)
  endif()
endif()
add_executable(pusch_demo src/demo.cpp)
target_link_libraries(pusch_demo PRIVATE pusch_lab)
add_executable(pusch_tests tests/test_main.cpp)
target_link_libraries(pusch_tests PRIVATE pusch_lab)
add_executable(pusch_bench src/bench.cpp)
add_executable(pusch_sweep src/sweep.cpp)
target_link_libraries(pusch_sweep PRIVATE pusch_lab)
enable_testing()
add_test(NAME pusch_algorithms COMMAND pusch_tests)
