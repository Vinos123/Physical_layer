cmake_minimum_required(VERSION 3.20)
project(nr_pusch_tutorial LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ENABLE_SANITIZERS "Enable address and undefined-behavior checks" OFF)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  add_compile_options(-Wall -Wextra -Wpedantic)
  if(ENABLE_SANITIZERS)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
  endif()
endif()
add_library(nr_core examples/nr_core.cpp)
target_include_directories(nr_core PUBLIC examples)
add_executable(nr_examples examples/nr_examples.cpp)
target_link_libraries(nr_examples PRIVATE nr_core)
add_executable(nr_tests examples/nr_tests.cpp)
target_link_libraries(nr_tests PRIVATE nr_core)
enable_testing()
add_test(NAME numerical_checks COMMAND nr_tests)
add_test(NAME experiments COMMAND nr_examples --out "${CMAKE_CURRENT_BINARY_DIR}/results")
